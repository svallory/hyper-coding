# Vite Integration

The HyperDev Vite plugin hooks into your build pipeline so template validation, type generation, and hot reloading all happen automatically --- without extra scripts or manual steps.

## Vite Plugin Configuration

Drop the plugin into your `vite.config.js`:

```javascript
// vite.config.js
import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'
import { hyperdevPlugin } from '@hypercli/vite-plugin'

export default defineConfig({
  plugins: [
    react(),
    hyperdevPlugin({
      // Auto-generate types on schema changes
      autoGenerateTypes: true,
      // Watch template changes in development
      watchTemplates: true,
      // Template directories to watch
      templateDirs: ['./.hyperdev/templates'],
      // Quality gates integration
      qualityGates: {
        enabled: true,
        failOnError: process.env.NODE_ENV === 'production',
        rules: ['typescript', 'eslint', 'security']
      },
      // Development features
      dev: {
        enableHMR: true,
        templateReload: true
      }
    })
  ],
  // HyperDev file processing
  define: {
    __HYPERDEV_TEMPLATES__: JSON.stringify(process.env.HYPERDEV_TEMPLATES)
  }
})
```

## Development Server Integration

Configure the Vite dev server to proxy HyperDev's dashboard and enable live reloading:

```javascript
// vite.config.js
export default defineConfig({
  server: {
    // Proxy HyperDev dashboard
    proxy: {
      '/hyperdev-dashboard': {
        target: 'http://localhost:4000',
        changeOrigin: true,
        rewrite: (path) => path.replace(/^\/hyperdev-dashboard/, '')
      }
    }
  },
  plugins: [
    hyperdevPlugin({
      dev: {
        // Enable live template reloading
        liveReload: true,
        // Watch for config changes
        watchConfig: true,
        // Auto-validate on template changes
        validateOnChange: true
      }
    })
  ]
})
```

## Build Integration

Run validation before every build so bad templates never make it into production:

```javascript
// vite.config.js
export default defineConfig({
  build: {
    rollupOptions: {
      plugins: [
        // Pre-build validation
        {
          name: 'hyperdev-pre-build',
          buildStart() {
            const { execSync } = require('child_process')
            try {
              execSync('hyper recipe list', { stdio: 'inherit' })
            } catch (error) {
              throw new Error('HyperDev not available — check installation')
            }
          }
        }
      ]
    }
  }
})
```

## Environment Configuration

Expose HyperDev config to the client through Vite's `define`:

```javascript
// vite.config.js
export default defineConfig({
  define: {
    // Make HyperDev config available to client
    __HYPERDEV_CONFIG__: JSON.stringify({
      templates: process.env.HYPERDEV_TEMPLATES,
      environment: process.env.NODE_ENV
    })
  },
  plugins: [
    hyperdevPlugin({
      // Environment-specific configuration
      templates: {
        development: {
          includeDebugInfo: true,
          verbose: true
        },
        production: {
          optimize: true,
          stripComments: true
        }
      }
    })
  ]
})
```

## TypeScript Integration

Auto-generate types from your templates and include them in the TypeScript compilation:

```javascript
// vite.config.js
export default defineConfig({
  plugins: [
    hyperdevPlugin({
      typescript: {
        // Auto-generate types from templates
        generateTypes: true,
        // Output directory for generated types
        typesOutput: './src/types/generated',
        // Watch schema files for changes
        watchSchemas: true,
        // Include in TypeScript compilation
        include: true
      }
    })
  ]
})
```

## Hot Module Replacement

Enable HMR for templates so changes reflect instantly in the browser:

```javascript
// vite.config.js
export default defineConfig({
  plugins: [
    hyperdevPlugin({
      hmr: {
        // Enable template hot reload
        templates: true,
        // Reload on config changes
        config: true,
        // Update types on schema changes
        types: true
      }
    })
  ]
})
```

Once this is set up, type generation and hot reloading become invisible parts of the build process --- they just happen.
