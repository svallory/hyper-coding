# Next.js Integration

HyperDev plugs into Next.js so you can generate API routes, components, and pages from templates — and keep generated code in sync as your templates evolve.

## Next.js Plugin Configuration

Set up the HyperDev Next.js plugin:

```javascript
// next.config.js
const { withHyperDev } = require('@hypercli/next-plugin')

/** @type {import('next').NextConfig} */
const nextConfig = {
  // ... other config
  experimental: {
    serverComponentsExternalPackages: ['@hypercli/core']
  }
}

module.exports = withHyperDev({
  // HyperDev configuration
  hyperdev: {
    // API route generation
    api: {
      autoGenerate: true,
      pattern: 'pages/api/**/*.template.yml',
      typescript: true
    },
    // Component generation
    components: {
      autoGenerate: true,
      pattern: 'components/**/*.template.yml',
      defaultProps: true
    },
    // Page generation
    pages: {
      autoGenerate: true,
      pattern: 'pages/**/*.template.yml',
      seo: true
    },
    // Quality gates
    qualityGates: {
      enabled: true,
      onBuild: true,
      rules: ['next-seo', 'accessibility', 'performance']
    }
  }
})(nextConfig)
```

## API Routes Integration

Generate Next.js API routes with HyperDev:

```yaml
# templates/api-route.template.yml
name: "Next.js API Route"
description: "Generate a Next.js API route with validation"
variables:
  - name: routeName
    description: "API route name"
    type: string
  - name: method
    description: "HTTP method"
    type: enum
    options: ["GET", "POST", "PUT", "DELETE"]
```

```typescript
// templates/api-route.jig
---
to: pages/api/{{ routeName }}.ts
---
import type { NextApiRequest, NextApiResponse } from 'next'

type Data = {
  message: string
}

export default function handler(
  req: NextApiRequest,
  res: NextApiResponse<Data>
) {
  if (req.method === '{{ method }}') {
    res.status(200).json({ message: 'Success' })
  } else {
    res.setHeader('Allow', ['{{ method }}'])
    res.status(405).end(`Method ${req.method} Not Allowed`)
  }
}
```

## Component Generation

Generate Next.js components with TypeScript:

```yaml
# templates/component.template.yml
name: "Next.js Component"
description: "Generate a Next.js React component"
variables:
  - name: componentName
    description: "Component name"
    type: string
  - name: withProps
    description: "Include props interface"
    type: boolean
    default: true
```

```tsx
// templates/component.jig
---
to: components/{{ componentName }}.tsx
---
@if(withProps)
interface {{ componentName }}Props {
  className?: string
}
@end

export default function {{ componentName }}(@if(withProps)props: {{ componentName }}Props@end) {
  return (
    <div@if(withProps) className={props.className}@end>
      <h1>{{ componentName }}</h1>
    </div>
  )
}
```

## Page Generation

Generate Next.js pages with SEO:

```tsx
// templates/page.jig
---
to: pages/{{ pageName }}.tsx
---
import Head from 'next/head'
import type { NextPage } from 'next'

const {{ pascalCase :: pageName }}Page: NextPage = () => {
  return (
    <>
      <Head>
        <title>{{ titleCase :: pageName }}</title>
        <meta name="description" content="{{ description || 'Generated page' }}" />
      </Head>
      
      <main>
        <h1>{{ titleCase :: pageName }}</h1>
        <p>This page was generated by HyperDev.</p>
      </main>
    </>
  )
}

export default {{ pascalCase :: pageName }}Page
```

## Development Scripts

Package.json scripts for Next.js and HyperDev:

```json
{
  "scripts": {
    "dev": "concurrently \"hyper watch\" \"next dev\"",
    "build": "next build",
    "start": "next start",
    "gen:api": "hyper gen api-route",
    "gen:component": "hyper gen component",
    "gen:page": "hyper gen page"
  }
}
```

## TypeScript Configuration

Configure TypeScript for HyperDev and Next.js:

```json
// tsconfig.json
{
  "compilerOptions": {
    // ... other options
    "paths": {
      "@/components/*": ["./components/*"],
      "@/pages/*": ["./pages/*"],
      "@/generated/*": ["./generated/*"]
    }
  },
  "include": [
    "next-env.d.ts",
    "**/*.ts",
    "**/*.tsx",
    ".hyperdev/generated/**/*"
  ]
}
```

With that config in place, `hyper watch` runs alongside `next dev` — templates are watched for changes, and generated files stay in sync.
