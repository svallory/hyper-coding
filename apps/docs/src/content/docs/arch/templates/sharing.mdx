## hyperdev-package.config.js Schema

```javascript
// hyperdev-package.config.js
module.exports = {
  name: '@myorg/react-templates',
  description: 'Professional React component templates',
  version: '2.1.0',
  author: 'Your Organization <dev@yourorg.com>',

  // Template registry — maps names to directories
  templates: {
    'component': './templates/component',
    'hook': './templates/hook',
    'service': './templates/service',
    'page': './templates/page',
    'workflow': './templates/workflow'
  },

  // Dependencies required by generated code
  peerDependencies: {
    'react': '^18.0.0',
    'typescript': '^4.9.0',
    '@types/react': '^18.0.0'
  },

  devDependencies: {
    '@hyperdev/action-system': '^2.0.0',
    'jest': '^29.0.0'
  },

  keywords: ['react', 'typescript', 'components', 'hooks'],
  category: 'frontend',
  license: 'MIT',
  homepage: 'https://github.com/myorg/react-templates',

  publishConfig: {
    registry: 'https://registry.npmjs.org',
    access: 'public'
  },

  engines: {
    node: '>=16.0.0',
    hyperdev: '^8.0.0'
  },

  trusted: true,
  maintainers: [
    'dev@yourorg.com',
    'security@yourorg.com'
  ]
};
```

## package.json Integration Fields

The `hyperdev` key in `package.json` signals that this package contains HyperDev templates:

```json
{
  "name": "@myorg/react-templates",
  "version": "2.1.0",
  "main": "index.js",
  "files": [
    "templates/**/*",
    "actions/**/*",
    "docs/**/*",
    "hyperdev-package.config.js",
    "README.md",
    "CHANGELOG.md"
  ],
  "keywords": ["hyperdev", "template", "react", "typescript", "generator"],
  "hyperdev": {
    "packageType": "templates",
    "version": "^8.0.0"
  },
  "peerDependencies": {
    "react": "^18.0.0",
    "typescript": "^4.9.0"
  },
  "scripts": {
    "test": "jest",
    "validate": "hyper validate-package",
    "build": "hyper build-package",
    "prepublishOnly": "bun run validate && bun test"
  }
}
```

## Repository-Level Discovery Config

```yaml
# .hyperdev/config.yml
repository:
  type: "template-collection"
  version: "2.1.0"

templates:
  discoveryPaths:
    - "templates/**"

  metadata:
    category: "react"
    maintainer: "dev@myorg.com"
    license: "MIT"

distribution:
  allowDirectInstall: true
  requireAuthentication: false
  trustLevel: "verified"

quality:
  requireTests: true
  enforceStandards: true
  minimumCoverage: 80

automation:
  validateOnPush: true
  autoRelease: true
  releasePattern: "v*"
```

## Automated Publishing Workflow

```yaml
# .github/workflows/release.yml
name: Release Templates
on:
  push:
    tags: ['v*']

jobs:
  validate:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: oven-sh/setup-bun@v1
      - run: bun install
      - run: bunx hyper validate-package
      - run: bun test

  publish:
    needs: validate
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: oven-sh/setup-bun@v1
      - name: Configure NPM
        run: echo "//registry.npmjs.org/:_authToken=${{ secrets.NPM_TOKEN }}" > ~/.npmrc
      - run: bun publish
```
